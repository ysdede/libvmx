cmake_minimum_required(VERSION 3.16)
project(VmxPreviewTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
enable_testing()
add_executable(vmx_preview preview.cpp ../src/vmxcodec.cpp)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
    target_sources(vmx_preview PRIVATE ../src/vmxcodec_arm.cpp)
else()
    target_sources(vmx_preview PRIVATE ../src/vmxcodec_x86.cpp ../src/vmxcodec_avx2.cpp)
    if(MSVC)
        target_compile_options(vmx_preview PRIVATE /arch:AVX2)
    else()
        target_compile_options(vmx_preview PRIVATE -march=haswell -fno-strict-aliasing)
    endif()
endif()
target_link_libraries(vmx_preview PRIVATE Threads::Threads)
add_test(NAME preview COMMAND vmx_preview)
set_tests_properties(preview PROPERTIES TIMEOUT 30)
